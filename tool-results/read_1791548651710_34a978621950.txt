    61→    61→function Waveform({ active, bars = 28, className }: { active: boolean; bars?: number; className?: string }) {
    62→    62→  return (
    63→    63→    <div className={cn("flex items-end justify-center gap-[3px] h-12", className)} aria-hidden="true">
    64→    64→      {Array.from({ length: bars }).map((_, i) => {
    65→    65→        const h = 22 + Math.abs(Math.sin(i * 0.9)) * 60 + (i % 4) * 6;
    66→    66→        return (
    67→    67→          <span
    68→    68→            key={i}
    69→    69→            className={cn("w-[4px] rounded-full brand-gradient", active && "wave-bar")}
    70→    70→            style={{
    71→    71→              height: `${Math.min(h, 100)}%`,
    72→    72→              animationDelay: `${(i % 9) * 0.09}s`,
    73→    73→              opacity: active ? 1 : 0.25,
    74→    74→            }}
    75→    75→          />
    76→    76→        );
    77→    77→      })}
    78→    78→    </div>
    79→    79→  );
    80→    80→}
