    25→  const postCount = "stats" in p ? p.stats.postCount : 0;
    26→  return [
    27→    { id: "join", emoji: "🎬", name: "AniVerse Citizen", desc: "Joined AniVerse", earned: true },
    28→    { id: "first", emoji: "💎", name: "First Purchase", desc: "Bought 1 anime", earned: purchaseCount >= 1 },
    29→    { id: "collector", emoji: "📚", name: "Collector", desc: "Owns 10 animes", earned: purchaseCount >= 10 },
    30→    { id: "exchange", emoji: "🔄", name: "Exchange Master", desc: "Owns 20 animes & trades", earned: purchaseCount >= 20 },
    31→    { id: "binge", emoji: "🍿", name: "Binge Watcher", desc: "5+ hours watched", earned: watchSeconds >= 5 * 3600 },
    32→    { id: "star", emoji: "⭐", name: "Community Star", desc: "Posted 5 times", earned: postCount >= 5 },
    33→    { id: "creator", emoji: "👑", name: "Creator", desc: "Publishes animes", earned: isCreator },

   304→/* ============ MOBILE / FULL PAGE PROFILE ============ */
   305→export function ProfileFullView({ username }: { username: string }) {
   306→  const { user, setAuthOpen, navigate } = useAppStore();
   307→  const [profile, setProfile] = useState<PublicProfile | null>(null);
   308→  const [loading, setLoading] = useState(true);
   309→  const [editOpen, setEditOpen] = useState(false);
   310→  const [favBusy, setFavBusy] = useState(false);
   311→
   312→  const isMe = user && profile && user.username === profile.username;
   313→
   314→  useEffect(() => {
   315→    setLoading(true);
   316→    fetch(`/api/users/${username}`)
   317→      .then((r) => r.json())
   318→      .then((d) => setProfile(d.profile || null))
   319→      .catch(() => toast.error("Failed to load profile"))
   320→      .finally(() => setLoading(false));
   321→  }, [username, user]);
   322→
   323→  async function toggleFollow() {
   324→    if (!user) {
   325→      setAuthOpen(true);
   326→      return;
   327→    }
   328→    if (!profile || isMe) return;
   329→    setFavBusy(true);
   330→    try {
   331→      const res = await fetch(`/api/users/${profile.username}/favorite`, { method: "POST" });
   332→      const d = await res.json();
   333→      if (res.ok) {
   334→        setProfile({
   335→          ...profile,
   336→          isFollowing: d.following,
   337→          stats: { ...profile.stats, followers: profile.stats.followers + (d.following ? 1 : -1) },
   338→        });
   339→        toast.success(d.following ? `Added ${profile.displayName} to favorites ♥` : "Removed from favorites");
   340→      }
   341→    } finally {
   342→      setFavBusy(false);
   343→    }
   344→  }
   345→
   346→  if (loading) {
   347→    return (
   348→      <div className="max-w-[760px] mx-auto px-4 py-6">
   349→        <div className="h-40 rounded-3xl skeleton mb-4" />
   350→        <div className="flex gap-4 items-end -mt-12 mb-4">
   351→          <div className="w-24 h-24 rounded-full skeleton" />
   352→          <div className="flex-1 space-y-2 pb-2">
   353→            <div className="h-5 w-40 rounded skeleton" />
   354→            <div className="h-3 w-28 rounded skeleton" />
   355→          </div>
   356→        </div>
   357→        <div className="h-24 rounded-2xl skeleton" />
   358→      </div>
   359→    );
   360→  }
   361→
   362→  if (!profile) {
   363→    return (
   364→      <div className="text-center py-32 px-4">
   365→        <p className="text-6xl mb-4">👻</p>
   366→        <p className="font-bold text-lg">User not found</p>
   367→        <Button variant="outline" className="mt-4 rounded-full" onClick={() => navigate("home")}>Back to Home</Button>
   368→      </div>
   369→    );
   370→  }
   371→
   372→  return (
   373→    <div className="max-w-[760px] mx-auto px-4 sm:px-8 py-4 sm:py-6 pb-10">
   374→      {/* Banner + avatar */}
   375→      <div className="relative banner-shine rounded-3xl overflow-hidden">
   376→        <div className="h-40 sm:h-56">
   377→          <SmartImage
   378→            src={profile.bannerUrl || profile.avatarUrl}
   379→            alt="Profile banner"
   380→            className="w-full h-full object-cover"
   381→            accentFrom={profile.accentColor}
   382→            accentTo="#1e1b4b"
   383→          />
   384→        </div>
   385→        <div className="absolute inset-0 bg-gradient-to-t from-background/80 via-transparent to-transparent" />
   386→      </div>
   387→
   388→      <div className="flex items-end gap-4 -mt-12 sm:-mt-14 px-2">
   389→        <div className="rounded-full p-1 bg-background shrink-0 shadow-xl">
   390→          <Avatar
   391→            src={profile.avatarUrl}
   392→            name={profile.displayName || profile.username}
   393→            accent={profile.accentColor}
   394→            size={96}
   395→          />
   396→        </div>
   397→        <div className="flex-1 min-w-0 pb-1">
   398→          <h1 className="text-xl sm:text-2xl font-black truncate">{profile.displayName || profile.username}</h1>
   399→          <p className="text-xs text-muted-foreground">@{profile.username}</p>
   400→        </div>
   401→      </div>
   402→
   403→      <div className="flex items-center gap-2 mt-4 px-1 flex-wrap">
   404→        <ProfileBadges p={profile} />
   405→        <span className="text-[11px] text-muted-foreground flex items-center gap-1">
   406→          <CalendarDays className="w-3 h-3" />
   407→          Joined {new Date(profile.createdAt).toLocaleDateString("en-US", { month: "short", year: "numeric" })}
   408→        </span>
   409→      </div>
   410→
   411→      {/* Actions */}
   412→      <div className="flex items-center gap-2.5 mt-4 flex-wrap">
   413→        {isMe ? (
   414→          <>
   415→            <Button className="rounded-full brand-gradient text-white font-bold" onClick={() => setEditOpen(true)}>
   416→              <Pencil className="w-4 h-4" /> Customize Profile

   439→      {/* Body */}

   444→              <Smile className="w-3.5 h-3.5" /> About me

   449→
   450→        <section className="glass rounded-2xl p-4 sm:p-5">
   451→          <p className="text-[10px] font-black uppercase tracking-wider text-muted-foreground mb-3">Stats</p>

   510→            <div className="grid grid-cols-1 sm:grid-cols-2 gap-2">
   511→              {profile.favoriteUsers.map((u) => (

   527→          )}
   528→        </section>
   529→
   530→        {/* Followers */}
   531→        {profile.followersList.length > 0 && (
   532→          <section className="glass rounded-2xl p-4 sm:p-5">

   544→                >
   545→                  <Avatar src={u.avatarUrl} name={u.displayName} accent={u.accentColor} size={22} />
   546→                  <span className="text-xs font-semibold">{u.displayName || u.username}</span>

   548→              ))}
   549→            </div>
   550→          </section>
   551→        )}
   552→
   553→        <div className="rounded-2xl border border-dashed p-4 flex items-start gap-2.5">
   554→          <MessageSquareOff className="w-4 h-4 text-muted-foreground shrink-0 mt-0.5" />

   556→            AniVerse keeps things friendly — no DMs. Interact through community posts, watch parties and favorites.
   557→          </p>
   558→        </div>
   559→      </div>
   560→
   561→      {isMe && user && <EditProfileDialog open={editOpen} onOpenChange={setEditOpen} onSaved={setProfile} me={user} />}
   562→    </div>
   563→  );
   564→}

   566→/* ============ EDIT PROFILE DIALOG ============ */
   567→export function EditProfileDialog({
   568→  open, onOpenChange, onSaved, me,
   569→}: {
   570→  open: boolean;
   571→  onOpenChange: (open: boolean) => void;
   572→  onSaved: (p: PublicProfile) => void;
   573→  me?: AuthUser;
   574→}) {
   575→  const { user, updateUser } = useAppStore();
   576→  const base = me || user;
   577→  const [displayName, setDisplayName] = useState("");
   578→  const [bio, setBio] = useState("");
   579→  const [avatarUrl, setAvatarUrl] = useState("");
   580→  const [bannerUrl, setBannerUrl] = useState("");
   581→  const [accent, setAccent] = useState("#a855f7");
   582→  const [tab, setTab] = useState<"avatar" | "banner" | "color">("avatar");
   583→  const [saving, setSaving] = useState(false);
   584→
   585→  useEffect(() => {
   586→    if (open && base) {
   587→      setDisplayName(base.displayName || "");
   588→      setBio(base.bio || "");
   589→      setAvatarUrl(base.avatarUrl || "");
   590→      setBannerUrl(base.bannerUrl || "");
   591→      setAccent(base.accentColor || "#a855f7");
   592→    }
   593→  }, [open, base]);
   594→
   595→  async function save() {
   596→    if (!displayName.trim()) {
   597→      toast.error("Display name cannot be empty");
   598→      return;
   599→    }
   600→    setSaving(true);
   601→    try {
   602→      const res = await fetch("/api/profile", {
   603→        method: "PATCH",
   604→        headers: { "Content-Type": "application/json" },
   605→        body: JSON.stringify({ displayName: displayName.trim(), bio, avatarUrl, bannerUrl, accentColor: accent }),
   606→      });
   607→      const d = await res.json();
   608→      if (!res.ok) {
   609→        toast.error(d.error || "Failed to save");
   610→        return;
   611→      }
   612→      updateUser(d.user);
   613→      toast.success("Profile updated ✨");
   614→      onOpenChange(false);
   615→      // Refresh profile data
   616→      fetch(`/api/users/${d.user.username}`)
   617→        .then((r) => r.json())
   618→        .then((pd) => pd.profile && onSaved(pd.profile))
   619→        .catch(() => {});
   620→    } catch {
   621→      toast.error("Network error");
   622→    } finally {
   623→      setSaving(false);
   624→    }
   625→  }
   626→
   627→  if (!base) return null;
   628→
   629→  return (
   630→    <Dialog open={open} onOpenChange={(o) => !saving && onOpenChange(o)}>
   631→      <DialogContent className="sm:max-w-[520px] rounded-2xl max-h-[88vh] overflow-y-auto fancy-scroll">
   632→        <DialogHeader>
   633→          <DialogTitle className="flex items-center gap-2">
   634→            <Pencil className="w-4.5 h-4.5 text-primary" /> Customize Profile
   635→          </DialogTitle>
   636→          <DialogDescription>Make it yours — banner, avatar, accent color and bio</DialogDescription>
   637→        </DialogHeader>
   638→
   639→        {/* Live preview */}
   640→        <div className="rounded-2xl overflow-hidden border">
   641→          <div className="h-24 relative banner-shine">
   642→            <SmartImage
   643→              src={bannerUrl || avatarUrl}
   644→              alt="banner preview"
   645→              className="w-full h-full object-cover"
   646→              accentFrom={accent}
   647→              accentTo="#1e1b4b"
   648→            />
   649→          </div>
   650→          <div className="px-4 pb-4 -mt-7 relative">
   651→            <div className="rounded-full p-1 inline-block" style={{ background: "var(--card)" }}>
   652→              <Avatar src={avatarUrl} name={displayName || base.username} accent={accent} size={56} />
   653→            </div>
   654→            <p className="font-black mt-1.5">{displayName || base.username}</p>
   655→            <div className="flex items-center gap-2 mt-1">
   656→              <span
   657→                className="w-3 h-3 rounded-full"
   658→                style={{ background: accent, boxShadow: `0 0 8px ${accent}` }}
   659→              />
   660→              <span className="text-[11px] text-muted-foreground">@{base.username} · {base.role.toLowerCase()}</span>
   661→            </div>
   662→          </div>
   663→        </div>
   664→
   665→        {/* Name & bio */}
   666→        <div className="space-y-3">
   667→          <div className="space-y-1.5">
   668→            <Label className="text-xs font-semibold">Display name</Label>
   669→            <Input
   670→              value={displayName}
   671→              onChange={(e) => setDisplayName(e.target.value)}
   672→              maxLength={32}
   673→              className="h-10 rounded-xl"
   674→            />
   675→          </div>
