    61→function Waveform({ active, bars = 28, className }: { active: boolean; bars?: number; className?: string }) {
    62→  return (
    63→    <div className={cn("flex items-end justify-center gap-[3px] h-12", className)} aria-hidden="true">
    64→      {Array.from({ length: bars }).map((_, i) => {
    65→        const h = 22 + Math.abs(Math.sin(i * 0.9)) * 60 + (i % 4) * 6;
    66→        return (
    67→          <span
    68→            key={i}
    69→            className={cn("w-[4px] rounded-full brand-gradient", active && "wave-bar")}
    70→            style={{
    71→              height: `${Math.min(h, 100)}%`,
    72→              animationDelay: `${(i % 9) * 0.09}s`,
    73→              opacity: active ? 1 : 0.25,
    74→            }}
    75→          />
    76→        );
    77→      })}
    78→    </div>
    79→  );
    80→}

    83→  return (

   128→
   129→export function DubbingView() {
   130→  const { user, navigate, updateUser } = useAppStore();
   131→
   132→  const [animes, setAnimes] = useState<AnimeSummary[] | null>(null);
   133→  const [query, setQuery] = useState("");
   134→  const [animeId, setAnimeId] = useState<string | null>(null);
   135→  const [episodes, setEpisodes] = useState<EpisodeOpt[] | null>(null);
   136→  const [epLoading, setEpLoading] = useState(false);
   137→  const [episodeId, setEpisodeId] = useState<string | null>(null);
   138→  const [voice, setVoice] = useState<string>("aziza");
   139→  const [jobs, setJobs] = useState<DubJobItem[]>([]);
   140→
   141→  // processing
   142→  const [stage, setStage] = useState(-1); // -1 idle, 0..4 stages, 5 done
   143→  const [submitting, setSubmitting] = useState(false);
   144→
   145→  const selectedAnime = animes?.find((a) => a.id === animeId) || null;
   146→  const selectedEpisode = episodes?.find((e) => e.id === episodeId) || null;
   147→  const selectedVoice = VOICES.find((v) => v.id === voice)!;
   148→
   149→  // load catalog (ownable: free or owned) + my jobs
   150→  useEffect(() => {
   151→    let alive = true;
   152→    fetch("/api/animes")
   153→      .then((r) => r.json())
   154→      .then((d) => {
   155→        if (!alive) return;
   156→        setAnimes((d.animes || []).filter((a: AnimeSummary) => a.isFree || a.owned));
   157→      })
   158→      .catch(() => toast.error("Failed to load animes"));
   159→    if (user) {
   160→      fetch("/api/dubbing")
   161→        .then((r) => r.json())
   162→        .then((d) => alive && setJobs(d.jobs || []))
   163→        .catch(() => {});
   164→    }
   165→    return () => {
   166→      alive = false;
   167→    };
   168→  }, [user?.id]);
   169→
   170→  // load episodes of the selected anime
   171→  useEffect(() => {
   172→    if (!animeId || !selectedAnime) {
   173→      setEpisodes(null);
   174→      return;
   175→    }
   176→    setEpLoading(true);
   177→    setEpisodeId(null);
   178→    fetch(`/api/animes/${selectedAnime.slug}`)
   179→      .then((r) => r.json())
   180→      .then((d) => setEpisodes(d.episodes || []))
   181→      .catch(() => toast.error("Failed to load episodes"))
   182→      .finally(() => setEpLoading(false));
   183→  }, [animeId, selectedAnime]);
   184→
   185→  const dubbedEpNumbers = useMemo(() => {
   186→    if (!selectedAnime) return new Set<number>();
   187→    return new Set(
   188→      jobs.filter((j) => j.animeSlug === selectedAnime.slug && j.targetLang === "uz").map((j) => j.episodeNumber)
   189→    );
   190→  }, [jobs, selectedAnime]);
   191→
   192→  const filtered = useMemo(() => {
   193→    const q = query.trim().toLowerCase();
   194→    if (!q) return animes || [];
   195→    return (animes || []).filter((a) => a.title.toLowerCase().includes(q) || a.genres.join(" ").toLowerCase().includes(q));
   196→  }, [animes, query]);
   197→
   198→  async function startDub() {
   199→    if (!user) {
   200→      navigate("login");
   201→      return;
   202→    }
   203→    if (!episodeId || !selectedAnime || !selectedEpisode) {
   204→      toast.error("Pick an anime and an episode first");
   205→      return;
   206→    }
   207→    if (user.credits < selectedVoice.price) {
   208→      toast.error(`Not enough credits — you need ${selectedVoice.price - user.credits} more`);
   209→      navigate("credits");
   210→      return;
   211→    }
   212→    setSubmitting(true);
   213→    setStage(0);
   214→    const startedAt = Date.now();
   215→
   216→    // stage choreography (min 4.4s) runs in parallel with the API call
   217→    const timers = [400, 1300, 2300, 3200, 4100].map((ms, i) =>
   218→      setTimeout(() => setStage(i + 1), ms)
   219→    );
   220→
   221→    try {
   222→      const res = await fetch("/api/dubbing", {
   223→        method: "POST",
   224→        headers: { "Content-Type": "application/json" },
   225→        body: JSON.stringify({ episodeId, voice }),
   226→      });
   227→      const data = await res.json();
   228→      if (!res.ok) {
   229→        toast.error(data.error || "Dubbing failed");
   230→        setStage(-1);
   231→        return;
   232→      }
   233→      // hold until the stage animation lands
   234→      const remaining = 4400 - (Date.now() - startedAt);
   235→      if (remaining > 0) await new Promise((r) => setTimeout(r, remaining));
   236→      setJobs((prev) => [data.job, ...prev]);
   237→      if (data.credits !== undefined) updateUser({ credits: data.credits });
   238→      toast.success(`EP${selectedEpisode.number} dubbed to Uzbek — ovozli tamilga tayyor!`);
   239→    } catch {
   240→      toast.error("Network error");
   241→      setStage(-1);
   242→    } finally {
   243→      timers.forEach(clearTimeout);
   244→      setSubmitting(false);
   245→    }
   246→  }
   247→
   248→  const processing = stage >= 0 && stage <= 4;
   249→
   250→  return (
   251→    <div className="pb-10">
   252→      {/* ===== Header ===== */}
   253→      <div className="max-w-[1280px] mx-auto px-4 sm:px-8 pt-8 pb-6">
   254→        <div className="flex items-end justify-between gap-4 flex-wrap">
   255→          <div>
   256→            <h1 className="font-display text-3xl sm:text-4xl font-black tracking-tight flex items-center gap-3">
   257→              <Logo variant="mark" size={40} state={processing ? "playing" : "idle"} animated />
   258→              AI Dubbing Studio
   259→            </h1>
   260→            <p className="text-muted-foreground text-sm mt-2 max-w-xl">
   261→              Replace Japanese voices with Uzbek — pick a series you own, choose a voice profile,
   262→              and let the pipeline do the rest. Powered by your credits.
   263→            </p>
   264→          </div>
   265→          {user && (
   266→            <div className="flex items-center gap-2 text-xs bg-card border rounded-full px-3.5 py-2 font-semibold">
   267→              <Coins className="w-3.5 h-3.5 text-amber-500" />
   268→              {formatCredits(user.credits)} credits
   269→            </div>
   270→          )}
   271→        </div>
   272→
   273→        {/* Language cards */}
   274→        <div className="grid grid-cols-1 sm:grid-cols-3 gap-3 mt-6">
   275→          {LANGUAGES.map((l) => (
   276→            <div
   277→              key={l.code}
   278→              className={cn(
   279→                "relative rounded-2xl border p-4 flex items-center gap-3.5 transition-all",
   280→                l.live
   281→                  ? "border-primary/50 bg-primary/[0.07] shadow-[0_0_24px_-12px_var(--glow)]"
   282→                  : "border-dashed opacity-60"
   283→              )}
   284→            >
   285→              <span className="text-2xl">{l.flag}</span>
   286→              <div className="flex-1">
   287→                <p className="font-bold text-[14px] flex items-center gap-2">
   288→                  {l.name}
   289→                  {l.live ? (
   290→                    <span className="text-[9px] font-black px-1.5 py-0.5 rounded-full bg-emerald-500/15 text-emerald-400 border border-emerald-500/30">
   291→                      LIVE
   292→                    </span>
   293→                  ) : (
   294→                    <Lock className="w-3 h-3 text-muted-foreground" />
   295→                  )}
   296→                </p>
   297→                <p className="text-[11px] text-muted-foreground mt-0.5">{l.note}</p>
   298→              </div>
   299→              {l.live && <AudioLines className="w-4 h-4 text-primary" />}
   300→            </div>
   301→          ))}
   302→        </div>
   303→      </div>
   304→
   305→      <div className="max-w-[1280px] mx-auto px-4 sm:px-8 grid lg:grid-cols-[1fr_360px] gap-6">
   306→        {/* ===== LEFT: pickers ===== */}
   307→        <div className="space-y-6">
   308→          {/* Processing panel replaces pickers while running */}
   309→          {processing ? (
   310→            <div className="rounded-2xl border aurora relative overflow-hidden p-6 sm:p-8">
   311→              <div className="relative flex flex-col sm:flex-row items-center gap-8">
   312→                <div className="flex-1 w-full">
   313→                  <p className="text-[11px] font-black uppercase tracking-[0.16em] text-primary mb-1 flex items-center gap-2">
   314→                    <Loader2 className="w-3.5 h-3.5 animate-spin" />
   315→                    Dubbing in progress
   316→                  </p>
   317→                  <h3 className="font-display text-xl font-black mb-4">
   318→                    {selectedAnime?.title} — EP{selectedEpisode?.number}
   319→                  </h3>
   320→                  <StageStepper current={stage} />
   321→                </div>
   322→                <div className="w-full sm:w-[300px] shrink-0">
   323→                  <div className="rounded-xl border bg-card/70 backdrop-blur p-5">
   324→                    <Waveform active bars={32} />
   325→                    <p className="text-center text-[11px] text-muted-foreground mt-3">
   326→                      日本語 → O'zbek · {selectedVoice.name} voice
   327→                    </p>
   328→                  </div>
   329→                </div>
   330→              </div>
   331→            </div>
   332→          ) : (
   333→            <>
   334→              {/* Anime picker */}
   335→              <section aria-label="Pick an anime">
   336→                <div className="flex items-center justify-between gap-3 mb-3">
   337→                  <h2 className="text-base font-bold flex items-center gap-2">
   338→                    <Film className="w-[18px] h-[18px] text-primary" />
   339→                    1 · Pick an anime
   340→                    <span className="text-[11px] font-medium text-muted-foreground">(free or owned)</span>
   341→                  </h2>
   342→                  <div className="relative w-52 sm:w-64">
   343→                    <Search className="w-4 h-4 absolute left-3 top-1/2 -translate-y-1/2 text-muted-foreground" />
   344→                    <Input
   345→                      value={query}
   346→                      onChange={(e) => setQuery(e.target.value)}
   347→                      placeholder="Filter..."
   348→                      className="pl-9 h-9 rounded-xl text-[13px]"
   349→                      aria-label="Filter animes"
   350→                    />
   351→                  </div>
   352→                </div>
   353→
   354→                {!animes ? (
   355→                  <div className="grid grid-cols-3 sm:grid-cols-4 xl:grid-cols-6 gap-3">
   356→                    {Array.from({ length: 6 }).map((_, i) => (
   357→                      <div key={i} className="aspect-[2/3] rounded-xl skeleton" />
   358→                    ))}
   359→                  </div>
   360→                ) : animes.length === 0 ? (
   361→                  <div className="rounded-2xl border border-dashed p-10 text-center">
   362→                    <Film className="w-9 h-9 mx-auto text-muted-foreground/50 mb-3" />
   363→                    <p className="font-bold">Nothing to dub yet</p>
   364→                    <p className="text-sm text-muted-foreground mt-1 mb-4">
   365→                      Dubbing works on free animes and titles in your library.
   366→                    </p>
   367→                    <Button size="sm" variant="outline" className="rounded-full" onClick={() => navigate("browse")}>
   368→                      Discover animes <ArrowRight className="w-4 h-4" />
   369→                    </Button>
   370→                  </div>
   371→                ) : (
   372→                  <div className="grid grid-cols-3 sm:grid-cols-4 xl:grid-cols-6 gap-3 max-h-[340px] overflow-y-auto pr-1">
   373→                    {filtered.map((a) => (
   374→                      <button
   375→                        key={a.id}
   376→                        onClick={() => setAnimeId(a.id)}
   377→                        className={cn(
   378→                          "group relative aspect-[2/3] rounded-xl overflow-hidden border-2 transition-all",
   379→                          animeId === a.id
   380→                            ? "border-primary shadow-[0_0_24px_-8px_var(--glow)]"
   381→                            : "border-transparent hover:border-primary/40"

   383→                        aria-pressed={animeId === a.id}

   394→                            <Check className="w-3 h-3 text-white" />

   399→                  </div>
   400→                )}
   401→              </section>
   402→
   403→              {/* Episode picker */}
   404→              {selectedAnime && (
   405→                <section aria-label="Pick an episode">
   406→                  <h2 className="text-base font-bold flex items-center gap-2 mb-3">
   407→                    <Play className="w-[18px] h-[18px] text-primary" />
   408→                    2 · Pick an episode
   409→                  </h2>
   410→                  {epLoading ? (
   411→                    <div className="space-y-2">
   412→                      {[0, 1, 2].map((i) => (
   413→                        <div key={i} className="h-12 rounded-xl skeleton" />
   414→                      ))}
   415→                    </div>

   417→                    <div className="grid sm:grid-cols-2 gap-2 max-h-[260px] overflow-y-auto pr-1">

   419→                        const dubbed = dubbedEpNumbers.has(ep.number);

   422→                            key={ep.id}
   423→                            disabled={dubbed}

   426→                              "flex items-center gap-3 rounded-xl border p-3 text-left transition-all",

   433→                            aria-pressed={episodeId === ep.id}
   434→                          >
   435→                            <span
   436→                              className={cn(
   437→                                "w-9 h-9 rounded-lg flex items-center justify-center text-[12px] font-black shrink-0",
   438→                                episodeId === ep.id ? "brand-gradient text-white" : "bg-accent"
   439→                              )}
   440→                            >
   441→                              {ep.number}

   443→                            <span className="flex-1 min-w-0">
   444→                              <span className="block text-[13px] font-bold truncate">{ep.title}</span>

   462→              )}
   463→            </>

   467→        {/* ===== RIGHT: voice profiles + order ===== */}
   468→        <div className="space-y-4">

   470→            <h2 className="text-base font-bold flex items-center gap-2 mb-1">
   471→              <Mic2 className="w-[18px] h-[18px] text-primary" />
   472→              3 · Voice profile
   473→            </h2>
   474→            <p className="text-[11px] text-muted-foreground mb-3">Uzbek voice cast — per episode pricing</p>
   475→            <div className="space-y-2">
   476→              {VOICES.map((v) => (
   477→                <button
   478→                  key={v.id}
   479→                  onClick={() => setVoice(v.id)}
   480→                  disabled={processing}
   481→                  className={cn(
   482→                    "w-full flex items-center gap-3 rounded-xl border p-2.5 text-left transition-all",
   483→                    voice === v.id
   484→                      ? "border-primary bg-primary/10"
   485→                      : "border-transparent hover:border-primary/30 bg-background/50"
   486→                  )}
   487→                  aria-pressed={voice === v.id}
   488→                >
   489→                  <span
   490→                    className={cn(
   491→                      "w-10 h-10 rounded-full bg-gradient-to-br shrink-0 flex items-center justify-center text-white font-black text-sm",
